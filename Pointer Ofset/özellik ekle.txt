using Address = uintptr_t;

static bool IsReadable(Address address, size_t size = sizeof(uint32_t))
{
    if (address + size < address)
        return false;

    MEMORY_BASIC_INFORMATION mbi{};
    if (VirtualQuery(reinterpret_cast<LPCVOID>(address), &mbi, sizeof(mbi)) != sizeof(mbi))
        return false;

    if (mbi.State != MEM_COMMIT)
        return false;

    if (mbi.Protect & (PAGE_NOACCESS | PAGE_GUARD))
        return false;

    Address regionBase = reinterpret_cast<Address>(mbi.BaseAddress);
    Address regionEnd  = regionBase + mbi.RegionSize;

    return address >= regionBase && (address + size) <= regionEnd;
}

template <typename T>
static T Read(Address address, T fallback = T{})
{
    static_assert(std::is_trivially_copyable<T>::value, "T trivially copyable olmalı");

    if (!IsReadable(address, sizeof(T)))
        return fallback;

    T value{};
    __try
    {
        std::memcpy(&value, reinterpret_cast<const void*>(address), sizeof(T));
    }
    __except (EXCEPTION_EXECUTE_HANDLER)
    {
        return fallback;
    }
    return value;
}

static bool IsExecutableAddress(Address address)
{
    MEMORY_BASIC_INFORMATION mbi{};
    if (VirtualQuery(reinterpret_cast<LPCVOID>(address), &mbi, sizeof(mbi)) != sizeof(mbi))
        return false;

    if (mbi.State != MEM_COMMIT)
        return false;

    if (mbi.Protect & (PAGE_NOACCESS | PAGE_GUARD))
        return false;

    const DWORD executableFlags =
        PAGE_EXECUTE | PAGE_EXECUTE_READ | PAGE_EXECUTE_READWRITE | PAGE_EXECUTE_WRITECOPY;

    return (mbi.Protect & executableFlags) != 0;
}

using UIClickFunction = void (__thiscall *)(Address self, uint32_t button, uint32_t flag);
#else
using UIClickFunction = void (*)(Address self, uint32_t button, uint32_t flag);

static Address GetMessageBox()
{
    Address manager = Read<Address>(KO_PTR_MSGBOX_MGR);
    if (!manager)
        return 0;

    return Read<Address>(manager + KO_OFF_AKTIFKUTU);
}

static bool ClickButton(Address button)
{
    Address parent = Read<Address>(button + KO_OFF_UIPARENT);
    if (!parent)
        return false;

    Address vtable = Read<Address>(parent);
    if (!vtable)
        return false;

    Address click = Read<Address>(vtable + KO_VT_UICLICK);
    if (!click || !IsExecutableAddress(click))
        return false;
  

    __try
    {
        UIClickFunction fn = reinterpret_cast<UIClickFunction>(click);
        fn(parent, button, 1);
    }
    __except (EXCEPTION_EXECUTE_HANDLER)
    {
        return false;
    }

    return true;
}

static bool PressButtonAt(Address buttonOffset)
{
    Address box = GetMessageBox();
    if (!box)
        return false;

    Address button = Read<Address>(box + buttonOffset);
    if (!button)
        return false;

    if (Read<Address>(button) != KO_VT_BUTTON)
        return false;

    return ClickButton(button);
}

1. 🏛 Oyun Belleğinin Yapısı: Pointer ve Offset Mantığı
Bir oyunun belleğinde (RAM) veriler rastgele dağınık değildir. Belirli bir düzen içinde, sabit bir "ana yapı" (genellikle struct veya class) üzerinden erişilebilirler. Knight Online'da karakterle ilgili tüm veriler (koordinatlar, can, mana, seviye, hareket durumu vb.) Karakter Yapısı (Character Structure) adı verilen bir bellek bölgesinde tutulur.

Pointer (İşaretçi): Bu yapının başlangıç adresini gösteren sabit bir referans noktasıdır. Örneğin, KO_PTR_CHR adlı pointer, karakter yapısının başladığı adresi tutar. Ancak bu adres, oyunun her açılışında (dynamic allocation nedeniyle) değişebilir . Bu yüzden asla sabit bir adres kullanılmaz; her seferinde pointer okunarak doğru adres bulunur.
Offset (Kayma Miktarı): Pointer'ın gösterdiği adresten itibaren, karakter yapısı içindeki belirli bir veriye (örneğin, hedef X koordinatı) ulaşmak için kaç byte ilerlenmesi gerektiğini belirten sabit sayıdır. Bu sayılar (KO_OFF_GoX, KO_OFF_MOVE gibi) oyun sürümü değişmedikçe genellikle aynı kalır .
Süreç: Bir veriye yazmak/okumak için önce KO_PTR_CHR adresini okur, sonra bu adrese ilgili offset'i ekleyerek verinin tam adresini (Dynamic Address) hesaplarsın. İşte verdiğin örnekteki Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_GoX") kodu tam olarak bunu yapar .
2. 🚶‍♂️ Adım Adım Memory ile Yürüme İşlemi
Verdiğin kod parçasını, arka planda gerçekleşen olaylarla birlikte detaylandıralım.

Adım 1: Hareket Tipini Belirleme (MOVEType)

cpp
Write4Byte(Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_MOVEType"), 2);
Ne Yapar? Karakterin hareket tipini 2 (koşma) olarak ayarlar.
Detayı: Bu, oyunun hareket motoruna "Yapılacak hareket bir koşma hareketidir" sinyalini verir. Eğer bu değer 1 ise yürüme, 0 ise durma gibi anlamlara gelebilir. Bu değer, hareketin hızını ve animasyonunu etkiler.
Adım 2: Hedef Koordinatları Belirleme (GoX, GoY)

cpp
WriteFloat(Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_GoX"), GoX);
WriteFloat(Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_GoY"), GoY);
Ne Yapar? Karakterin gitmesini istediğin hedefin X ve Y koordinatlarını float (ondalıklı sayı) formatında bellek adreslerine yazar.
Detayı: Bu değerler, oyunun "hedef noktası (waypoint)" olarak kabul ettiği değerlerdir. Koordinatların doğru ve hassas olması çok önemlidir. Aksi takdirde karakter hedefi bulamaz veya sapma yaşar.
Adım 3: Hareketi Başlatma (MOVE) - En Kritik Adım

cpp
Write4Byte(Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_MOVE"), 1);
Ne Yapar? Hareket mekanizmasını başlatan tetikleyiciyi (1) yazar.
Detayı: Bu adım, paket göndermedeki 0x06 paketini göndermekten çok daha güçlüdür. Bu bayrak 1 olarak ayarlandığında, oyunun kendi iç döngüsü (game loop) bu değişikliği algılar ve aşağıdaki süreci otomatik olarak başlatır:
MOVEType, GoX ve GoY değerlerini okur.
Karakterin mevcut pozisyonu ile hedef arasında bir yol (path) hesaplar (engelleri aşmak için pathfinding yapar).
Hesaplanan bu yol üzerindeki adımları (step'leri) ve yön bilgisini (heading) kendi iç buffer'ında oluşturur.
Her oyun tick'inde (genellikle saniyede 10-20 kez), buffer'daki bir sonraki adım koordinatını alarak, sunucuya gitmesi gereken doğal hareket paketlerini (0x06 ve benzeri) otomatik olarak oluşturur ve gönderir .
Karakter hedefe ulaştığında veya yeni bir komut verilene kadar bu işlem devam eder.
Bu sayede, paket gönderme yönteminde yaşadığın "küçük adım atma", "DC olma" veya "zip" hareketi gibi sorunlar yaşanmaz. Çünkü tüm karmaşık hareket mantığını sen yönetmeye çalışmazsın, oyunun kendi sağlam mekanizmasına emanet edersin.

3. 💡 Neden Bu Yöntem Daha Avantajlı?
Doğallık: Karakter, tıpkı bir oyuncu tarafından fare ile tıklanmış gibi doğal ve akıcı hareket eder. Yürüme, koşma ve durma animasyonları sorunsuz çalışır.
Kararlılık: Oyunun kendi hız sınırlamaları ve anti-cheat mekanizmaları (varsa) bu yöntemle daha uyumlu çalışır. Anormal paket göndermekten kaynaklanan DC'ler büyük ölçüde ortadan kalkar.
Kolaylık: Paket yapısını (0x06'nın yanındaki 1C 00, 2D 00 gibi anlam bilinmeyen byte'ları) çözmekle uğraşmazsın.
4. ⚠ Dikkat Edilmesi Gereken Hayati Noktalar
Güncel Offset ve Pointer'ların Bulunması: Bu yöntemin en büyük zorluğu, doğru adresleri bulmaktır. Oyun güncellendikçe veya farklı sunucularda (USKO, PVP Serverlar) pointer ve offset değerleri değişir . Bu değerleri bulmak için:
Cheat Engine: Bellek taraması yaparak değerlerin adreslerini bulabilirsin.
AOB (Array of Bytes) Taraması: Belli bir kod kalıbını (örneğin, hareket fonksiyonunun başlangıç byte'ları) bularak pointer'ları dinamik olarak tespit eden yöntemler kullanabilirsin .
Topluluk Paylaşımları: ko4life.net, elfdaily.com gibi forumlarda güncel offset paylaşımlarını takip edebilirsin . Ayrıca, açık kaynaklı projeler (örneğin OpenKO) bu değerleri anlamak için iyi bir referans olabilir .
Koordinatların Doğruluğu: Yazdığın GoX ve GoY değerlerinin oyunun kullandığı koordinat sistemiyle uyumlu olması şarttır. Genellikle bu değerler oyun içi harita koordinatlarıdır.
Zamanlama (Timing): KO_OFF_MOVE'a yazdıktan sonra, karakterin hedefe ulaşıp ulaşmadığını kontrol etmek için döngüye girmen gerekir. Aksi takdirde karakter hedefe vardığı halde "hareket et" komudu aktif kalabilir.
Senkronizasyon (Desync): Bellekte yazdığın hedef koordinat ile oyunun sunucudan gelen gerçek pozisyon verisi arasında fark oluşabilir . Bu nedenle, karakterin anlık pozisyonunu da bellekten okuyarak (ReadFloat(KO_PTR_CHR + KO_OFF_X)) hedefe yaklaşıp yaklaşmadığını takip etmen iyi olur.

Move tarafında kullandığım packet şu formatta:

06 [X*10:2] [Z*10:2] [1C 00] [2D 00] [03] [prevX*10:2] [prevZ*10:2] [1C 00]

Canlı hareket paketinden çıkardığım mantık şu şekilde:

- 06 = move opcode
- X/Z alanları little-endian ve coordinate * 10
- prevX/prevZ alanları da yine coordinate * 10
- 1C 00 sabiti mevcut
- 2D 00 ve 03 alanları da pakette geliyor

Örnek bir eşleşmede:

- packet içindeki 0x1D06 → 743.0
- packet içindeki 0x08DA → 226.6

ve bunlar karakterin gerçek pozisyonuyla birebir tuttu.

    int32_t totalCount = 0;

    DWORD gameProcMain = *(DWORD*)KO_PTR_GameProcMain;
    DWORD inventoryBase = *(DWORD*)(gameProcMain + KO_OFF_INN);

    for (int i = 0; i < HAVE_MAX_INN; i++)
    {
        DWORD itemPtr = *(DWORD*)(inventoryBase + KO_OFF_INN_FirstItem + (i * 4));

        if (itemPtr == 0)
        {
            printf("slot=%d (0x%02X), itemID=0 (LE: 00 00 00 00), count=0 (0x00)\n", i, i);
            continue;
        }

        // baseID pointer
        DWORD baseIDPtr = *(DWORD*)(itemPtr + 0x60);
        DWORD baseID = (baseIDPtr ? *(DWORD*)baseIDPtr : 0);

        // level pointer
        DWORD levelPtr = *(DWORD*)(itemPtr + 0x64);
        DWORD level = (levelPtr ? *(DWORD*)levelPtr : 0);

        // final ID = baseID + level
        DWORD finalID = baseID + level;

        // count
        DWORD count = *(DWORD*)(itemPtr + 0x68);

        // little-endian byte dizisi
        BYTE* b = (BYTE*)&finalID;

        printf(
            "slot=%d (0x%02X), baseID=%u, level=%u, finalID=%u (LE: %02X %02X %02X %02X), count=%u (0x%X)\n",
            i,                          // decimal slot
            i,                          // hex slot
            baseID,                     // base item ID
            level,                      // upgrade level
            finalID,                    // final ID
            b[0], b[1], b[2], b[3],     // LE byte dizisi
            count,                      // decimal count
            count                       // hex count
        );

        totalCount += count;
    }

    return totalCount;
}

//inn hostese transfer
void TransferLevel6ItemsToInn()
{
    DWORD gameProcMain = *(DWORD*)KO_PTR_GameProcMain;
    DWORD invBase = *(DWORD*)(gameProcMain + KO_OFF_UIInventory);
    DWORD innBase = *(DWORD*)(gameProcMain + KO_OFF_INN);

    struct ItemInfo { int slot; DWORD finalID; };
    std::vector<ItemInfo> items;

    for (int slot = 0; slot < HAVE_MAX; slot++)
    {
        DWORD itemPtr = *(DWORD*)(invBase + KO_OFF_UIInventory_FirstItem + (slot * 4));
        if (!itemPtr) continue;

        DWORD baseIDPtr = *(DWORD*)(itemPtr + 0x60);
        DWORD baseID = (baseIDPtr ? *(DWORD*)baseIDPtr : 0);

        DWORD levelPtr = *(DWORD*)(itemPtr + 0x64);
        DWORD level = (levelPtr ? *(DWORD*)levelPtr : 0);

        DWORD finalID = baseID + level;

        if (level == 7) { // items level
            items.push_back({ slot, finalID });
        }
    }


    std::vector<int> emptySlots;
    for (int j = 0; j < HAVE_MAX_INN; j++)
    {
        DWORD innItemPtr = *(DWORD*)(innBase + KO_OFF_INN_FirstItem + (j * 4));
        if (innItemPtr == 0) {
            emptySlots.push_back(j);
        }
    }

    size_t pairCount = min(items.size(), emptySlots.size());
    for (size_t i = 0; i < pairCount; i++)
    {
        int invSlot = items[i].slot;
        DWORD finalID = items[i].finalID;
        int innSlot = emptySlots[i];

        BYTE packet[17] = {
            0x45, 0x02, 0x2F, 0xF0, 0x00, 0x00,
            0,0,0,0, // itemID
            0x00,0,     // inventory slot
            0,0,     // inn slot
            0x00,0x00,0x00
        };

        BYTE* idBytes = (BYTE*)&finalID;
        packet[6] = idBytes[0];
        packet[7] = idBytes[1];
        packet[8] = idBytes[2];
        packet[9] = idBytes[3];


        packet[11] = (BYTE)(invSlot & 0xFF);

        packet[12] = (BYTE)(innSlot & 0xFF);  // ikisinden biri fazlal�k kontrol et bi ara
        packet[13] = (BYTE)((innSlot >> 8) & 0xFF);

        send(packet, sizeof(packet));
        Sleep(200);
    }
}

    for (int i = 0; i < HAVE_MAX; i++) {
        if (soldSlots >= HAVE_MAX) break;

        DWORD itemPtr = *(DWORD*)(inventoryBase + KO_OFF_UIInventory_FirstItem + (i * 4));
        if (!itemPtr) continue;

        DWORD itemBasic = *(DWORD*)(itemPtr + KO_OFF_INV_ITEM_ID);
        if (!itemBasic) continue;
        if (itemBasic == 810418000) continue; // filtrelenmi� ID

        DWORD baseID = *(DWORD*)itemBasic;
        if (baseID == 0) continue;

        DWORD count = *(DWORD*)(itemPtr + 0x68);
        if (count == 0) count = 1; // count 0 olsa bile en az 1 paket g�nder

        BYTE slotID = (BYTE)i;
        BYTE satilanAdet = 1;

        //sa�ma bi�ey yazd�m itemin ger�ek ID sinin tutuldu�u adres varsa g�rkem bilir
        for (int j = 0; j < 6; j++) {
            DWORD sendID = (baseID & 0xFFFFFF00) | (((baseID & 0xFF) + j) & 0xFF);
            sndSalePacket(satilanAdet, sendID, slotID, 0x01);
            Sleep(100);
        }

        soldSlots++;
    }

1) Mob ölür, kutu düşer → RECV 0x23 (BoxID öğrenilir)

└─ 10 byte = alınamadı (hata)

public void MoveCoordinate(int GoX, int GoY)
{
if (GoX <= 0 || GoY <= 0) return;
if (GoX == GetX() && GoY == GetY()) return;
Write4Byte(Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_MOVEType"), 2);
WriteFloat(Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_GoX"), GoX);
WriteFloat(Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_GoY"), GoY);
Write4Byte(Read4Byte(GetAddress("KO_PTR_CHR")) + GetAddress("KO_OFF_MOVE"), 1);
}


#define KO_OFF_GOX 0x1020
#define KO_OFF_GOY 0x1028
#define KO_OFF_MOVE 0x1014
#define KO_OFF_MOVETYPE 0x0140

{

{
*(DWORD*)(chr + KO_OFF_MOVETYPE) = 2;
*(float*)(chr + KO_OFF_MOUSE_X) = x;

*(BYTE*) (chr + KO_OFF_MOVE) = 1;
return true;
}
__except (EXCEPTION_EXECUTE_HANDLER)

return false;
}
}
