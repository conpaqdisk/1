            Hook::HookDirectX();
            return 0;

bool Hook::GetD3D9Device(void** pTable, size_t Size) {
    HWND hWnd = GetProcessWindow();
    if (!hWnd) return false;

    WNDCLASSEXA wc = { sizeof(WNDCLASSEXA), CS_CLASSDC, DefWindowProcA, 0L, 0L, GetModuleHandleA(nullptr), nullptr, nullptr, nullptr, nullptr, "DummyWindow", nullptr };
    RegisterClassExA(&wc);
    HWND hDummy = CreateWindowExA(0, "DummyWindow", nullptr, WS_OVERLAPPEDWINDOW, 0, 0, 100, 100, nullptr, nullptr, wc.hInstance, nullptr);
    if (!hDummy) return false;

    LPDIRECT3D9 pD3D = Direct3DCreate9(D3D_SDK_VERSION);
    if (!pD3D) { DestroyWindow(hDummy); return false; }

    D3DPRESENT_PARAMETERS pp = { 0 };
    pp.Windowed = TRUE;
    pp.SwapEffect = D3DSWAPEFFECT_DISCARD;
    pp.hDeviceWindow = hDummy;

    LPDIRECT3DDEVICE9 pDevice = nullptr;
    HRESULT hr = pD3D->CreateDevice(D3DADAPTER_DEFAULT, D3DDEVTYPE_HAL, hDummy, D3DCREATE_SOFTWARE_VERTEXPROCESSING, &pp, &pDevice);
    if (FAILED(hr)) { pD3D->Release(); DestroyWindow(hDummy); return false; }

    memcpy(pTable, *(void***)pDevice, Size);

    pDevice->Release();
    pD3D->Release();
    DestroyWindow(hDummy);
    return true;
}

HWND Hook::GetProcessWindow() {
    DWORD pid = GetCurrentProcessId();
    HWND hWnd = nullptr;
    do {
        hWnd = FindWindowEx(nullptr, hWnd, nullptr, nullptr);
        DWORD windowPid;
        GetWindowThreadProcessId(hWnd, &windowPid);
        if (pid == windowPid) {
            char className[256];
            GetClassNameA(hWnd, className, sizeof(className));
            if (strcmp(className, "KnightOnline") == 0) break;
        }
    } while (hWnd != nullptr);
    return hWnd;
}

void Hook::HookDirectX() {
    if (GetD3D9Device(d3d9Device, sizeof(d3d9Device))) {
        oEndScene = (tEndScene)d3d9Device[42];
        oReset = (tReset)d3d9Device[16];

        DetourTransactionBegin();
        DetourUpdateThread(GetCurrentThread());
        DetourAttach(&(PVOID&)oEndScene, Drawing::hkEndScene);
        DetourAttach(&(PVOID&)oReset, hkReset);
        DetourTransactionCommit();

        LOG_INFO("DirectX 9 hook başarıyla kuruldu.");
    }

    Commands::InitializePipeServer();
}

void Hook::UnHookDirectX() {
    if (Drawing::bInit) {
        UnHookWindow();
        ImGui_ImplDX9_Shutdown();
        ImGui_ImplWin32_Shutdown();
        ImGui::DestroyContext();
    }
    Drawing::bInit = false;

    Commands::ShutdownPipeServer();

    if (oEndScene && oReset) {
        DetourTransactionBegin();
        DetourUpdateThread(GetCurrentThread());
        DetourDetach(&(PVOID&)oEndScene, Drawing::hkEndScene);
        DetourDetach(&(PVOID&)oReset, hkReset);
        DetourTransactionCommit();
        LOG_INFO("DirectX 9 hook kaldırıldı.");
    }
}

void Hook::HookWindow() {
    if (!window) return;
    OWndProc = (WNDPROC)SetWindowLongPtr(window, GWLP_WNDPROC, (LONG_PTR)WndProc);
}

void Hook::UnHookWindow() {
    if (window && OWndProc) {
        SetWindowLongPtr(window, GWLP_WNDPROC, (LONG_PTR)OWndProc);
    }
}

LRESULT Hook::WndProc(HWND hWnd, UINT uMsg, WPARAM wParam, LPARAM lParam) {
    if (Drawing::bInit) {
        ImGui_ImplWin32_WndProcHandler(hWnd, uMsg, wParam, lParam);
    }
    return CallWindowProc(OWndProc, hWnd, uMsg, wParam, lParam);
}

HRESULT __stdcall hkReset(LPDIRECT3DDEVICE9 pDevice, D3DPRESENT_PARAMETERS* pPresentationParameters) {
    if (Drawing::bInit) {
        ImGui_ImplDX9_InvalidateDeviceObjects();
        HRESULT hr = Hook::oReset(pDevice, pPresentationParameters);
        ImGui_ImplDX9_CreateDeviceObjects();
        return hr;
    }
    return Hook::oReset(pDevice, pPresentationParameters);
}

static const wchar_t* PIPE_NAME = L"\\\\.\\pipe\\SteelBoxPipe";

void Commands::InitializePipeServer() {
    if (g_hPipe != INVALID_HANDLE_VALUE) return;
    g_hPipe = CreateNamedPipeW(
        PIPE_NAME,
        PIPE_ACCESS_DUPLEX,
        PIPE_TYPE_MESSAGE | PIPE_READMODE_MESSAGE | PIPE_WAIT,
        1, 1024, 1024, 0, nullptr
    );
    if (g_hPipe != INVALID_HANDLE_VALUE) {
        LOG_INFO("Named pipe sunucusu oluşturuldu.");
    }
}

void Commands::ShutdownPipeServer() {
    if (g_hPipe != INVALID_HANDLE_VALUE) {
        CloseHandle(g_hPipe);
        g_hPipe = INVALID_HANDLE_VALUE;
        LOG_INFO("Named pipe sunucusu kapatıldı.");
    }
}

std::string Commands::ReadPipe() {
    if (g_hPipe == INVALID_HANDLE_VALUE) return "";
    if (!ConnectNamedPipe(g_hPipe, nullptr)) {
        if (GetLastError() != ERROR_PIPE_CONNECTED) return "";
    }
    char buffer[1024] = {0};
    DWORD bytesRead;
    if (ReadFile(g_hPipe, buffer, sizeof(buffer) - 1, &bytesRead, nullptr)) {
        return std::string(buffer, bytesRead);
    }
    DisconnectNamedPipe(g_hPipe);
    return "";
}

void Commands::WritePipe(const std::string& response) {
    if (g_hPipe == INVALID_HANDLE_VALUE || response.empty()) return;
    DWORD bytesWritten;
    WriteFile(g_hPipe, response.c_str(), (DWORD)response.size(), &bytesWritten, nullptr);
    FlushFileBuffers(g_hPipe);
    DisconnectNamedPipe(g_hPipe);
}

std::string Commands::Process(const std::string& command) {
    if (command == "PING") {
        return "PONG";
    }
    if (command == "CHARINFO") {
        auto& mem = Memory::Get();
        auto chr = mem.GetChrPtr();
        if (chr) {
            std::stringstream ss;
            ss << "NAME:" << mem.GetCharacterName(chr) << "|LVL:" << mem.GetLevel(chr)
               << "|HP:" << mem.GetHP(chr) << "|MAXHP:" << mem.GetMaxHP(chr)
               << "|MP:" << mem.GetMP(chr) << "|MAXMP:" << mem.GetMaxMP(chr)
               << "|GOLD:" << mem.GetGold(chr);
            return ss.str();
        }
        return "ERROR:No character";
    }
    if (command == "TOWN") {
        // Town ışınlanma paketi gönder
        return "TOWN_OK";
    }
    if (command.find("LICENSE:") == 0) {
        std::string key = command.substr(8);
        bool valid = AppState::Get().ValidateLicense(key);
        return valid ? "LICENSE_OK" : "LICENSE_FAIL";
    }
    return "UNKNOWN_COMMAND";
}

class Memory {
public:
    static Memory& Get();

    bool AttachToGame();
    void Detach();

    template<typename T>
    T Read(uintptr_t address) {
        T value{};
        if (m_hProcess != INVALID_HANDLE_VALUE) {
            __try {
                SIZE_T bytesRead;
                ReadProcessMemory(m_hProcess, (LPCVOID)address, &value, sizeof(T), &bytesRead);
            } __except (EXCEPTION_EXECUTE_HANDLER) {
                return value;
            }
        }
        return value;
    }

    template<typename T>
    bool Write(uintptr_t address, const T& value) {
        if (m_hProcess == INVALID_HANDLE_VALUE) return false;
        SIZE_T bytesWritten;
        return WriteProcessMemory(m_hProcess, (LPVOID)address, &value, sizeof(T), &bytesWritten) == TRUE;
    }

    uintptr_t GetBaseAddress() const { return m_baseAddress; }
    uintptr_t GetChrPtr();
    uintptr_t GetPktPtr();

    int GetHP(uintptr_t chr);
    int GetMaxHP(uintptr_t chr);
    int GetMP(uintptr_t chr);
    int GetMaxMP(uintptr_t chr);
    int GetLevel(uintptr_t chr);
    int GetGold(uintptr_t chr);
    float GetPosX(uintptr_t chr);
    float GetPosY(uintptr_t chr);
    float GetPosZ(uintptr_t chr);
    std::string GetCharacterName(uintptr_t chr);

private:
    Memory() = default;
    ~Memory() { Detach(); }
    Memory(const Memory&) = delete;
    Memory& operator=(const Memory&) = delete;

    HANDLE m_hProcess = INVALID_HANDLE_VALUE;
    uintptr_t m_baseAddress = 0;
    DWORD m_processId = 0;
};

bool Memory::AttachToGame() {
    HANDLE snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snapshot == INVALID_HANDLE_VALUE) {
        LOG_ERROR("Process snapshot oluşturulamadı");
        return false;
    }

    PROCESSENTRY32W pe = { sizeof(PROCESSENTRY32W) };
    bool found = false;
    if (Process32FirstW(snapshot, &pe)) {
        do {
            if (wcscmp(pe.szExeFile, L"KnightOnLine.exe") == 0 ||
                wcscmp(pe.szExeFile, L"KnightOnline.exe") == 0) {
                m_processId = pe.th32ProcessID;
                found = true;
                break;
            }
        } while (Process32NextW(snapshot, &pe));
    }
    CloseHandle(snapshot);

    if (!found) {
        LOG_ERROR("KnightOnLine.exe bulunamadı!");
        return false;
    }

    m_hProcess = OpenProcess(PROCESS_ALL_ACCESS, FALSE, m_processId);
    if (m_hProcess == INVALID_HANDLE_VALUE) {
        LOG_ERROR("Proses açılamadı, yönetici yetkisi gerekli");
        return false;
    }

    HMODULE modules[1024];
    DWORD needed;
    if (EnumProcessModules(m_hProcess, modules, sizeof(modules), &needed)) {
        m_baseAddress = (uintptr_t)modules[0];
    } else {
        LOG_ERROR("Base adresi bulunamadı");
        Detach();
        return false;
    }

    LOG_INFO("Oyun sürecine bağlanıldı, PID: %d, Base: 0x%llX", m_processId, m_baseAddress);
    return true;
}

void Memory::Detach() {
    if (m_hProcess != INVALID_HANDLE_VALUE) {
        CloseHandle(m_hProcess);
        m_hProcess = INVALID_HANDLE_VALUE;
    }
    m_baseAddress = 0;
}

    std::lock_guard<std::mutex> lock(m_mutex);
    if (index >= 0 && index < (int)m_accounts.size()) {

    std::ifstream file("accounts.json");
    if (!file.is_open()) return;
    json j;
    file >> j;
    if (j.contains("accounts")) {
        for (auto& item : j["accounts"]) {
            Account acc;
            acc.username = item["username"];
            acc.passwordHash = item["password"];
            acc.server = item["server"];
            acc.platform = item["platform"];
            acc.characterName = item["charName"];
            acc.isActive = item["isActive"];
            acc.configId = item["configId"];
            m_accounts.push_back(acc);
        }
        LOG_INFO("%d hesap yüklendi.", (int)m_accounts.size());
    }
}

void AppState::SaveAccounts() {
    json j;
    json accountsArray = json::array();
    for (const auto& acc : m_accounts) {
        json item;
        item["username"] = acc.username;
        item["password"] = acc.passwordHash;
        item["server"] = acc.server;
        item["platform"] = acc.platform;
        item["charName"] = acc.characterName;
        item["isActive"] = acc.isActive;
        item["configId"] = acc.configId;
        accountsArray.push_back(item);
    }
    j["accounts"] = accountsArray;
    std::ofstream file("accounts.json");
    if (file.is_open()) {
        file << j.dump(4);
        LOG_INFO("Hesaplar kaydedildi.");
    }

void Logger::Info(const char* format, ...) {
    char buffer[1024];
    va_list args;
    va_start(args, format);
    vsnprintf(buffer, sizeof(buffer), format, args);
    va_end(args);
    Write("INFO", buffer);

    char buffer[1024];
    va_list args;
    va_start(args, format);
    vsnprintf(buffer, sizeof(buffer), format, args);
    va_end(args);

    char buffer[1024];
    va_list args;
    va_start(args, format);
    vsnprintf(buffer, sizeof(buffer), format, args);
    va_end(args);
    Write("ERROR", buffer);
}

void Logger::Write(const std::string& level, const std::string& message) {
    std::lock_guard<std::mutex> lock(m_mutex);
    if (!m_initialized) return;

    time_t now = time(nullptr);
    char timeStr[64];
    tm local;
    localtime_s(&local, &now);
    strftime(timeStr, sizeof(timeStr), "%Y-%m-%d %H:%M:%S", &local);

    std::string line = std::string("[") + timeStr + "] [" + level + "] " + message;
    if (m_file.is_open()) {
        m_file << line << std::endl;
        m_file.flush();
    }
    printf("%s\n", line.c_str());
}
