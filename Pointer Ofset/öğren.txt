VTable'ı ezdik, peki paket geldiğinde oyunu çökertmeden nasıl okuyacağız? Bunun için __declspec(naked) kullanarak hiçbir C++ derleyici kodu eklenmemiş saf bir Assembly bloğu oluşturuyoruz. Register'ları (yazmaçları) kaydedip kendi C++ analiz fonksiyonumuzu çağırıyoruz.

C++:
// Oyun bu adrese atladığında hiçbir register bozulmamalı!
__declspec(naked) void HookRecvStatic() {
    __asm {
        push ebp
        mov ebp, esp
        pushfd            // Tüm bayrakları (EFLAGS) yedekle
        pushad            // Tüm yazmaçları (EAX, EBX, ECX vs.) yedekle
       
        // C++ fonksiyonumuza (SafeProcessPacket) gönderilecek parametreleri yığına (stack) at
        push dword ptr [ebp+12] // esp8 (Argüman 2)
        push dword ptr [ebp+8]  // esp4 (Argüman 1)
        push edx                // Fastcall EDX
        push ecx                // Fastcall ECX / This
       
        // Paket analiz fonksiyonunu çağır
        call SafeProcessPacket
        add esp, 16             // Yığını (stack) temizle
       
        popad             // Yazmaçları eski haline getir
        popfd             // Bayrakları eski haline getir
        mov esp, ebp
        pop ebp
       
        // Sanki hiçbir şey olmamış gibi orijinal KO Recv fonksiyonuna zıpla (JMP)
        jmp dword ptr [g_origVFuncs + 8]
    }
}

Knight Online sürümlerinde paket sınıfı (packet class) sürekli yapı değiştirdiği için paketin uzunluğu (len) ve veri işareti (*pData) farklı yazmaçlarda (ECX, EDX) veya Yığın (Stack) üzerinde gezebilir. Bu nedenle her yeri arayan dinamik bir arayıcı (Scanner) yazdık:

bool TryProcessPacket(void* ptr) {
    if (!ptr) return false;
    __try {
        // Yeni Nesil KO için Offset yapılandırması
        uint32_t len1 = *(uint32_t*)((uint32_t)ptr + 0);
        uint8_t* pData1 = *(uint8_t**)((uint32_t)ptr + 4);
       
        // Mantıklı bir paket mi diye güvenlik kontrolleri (0 < len < 2000 ve adres geçerli mi)
        if (len1 > 0 && len1 < 2000 && pData1 && (uint32_t)pData1 > 0x10000) {
            uint8_t opcode = pData1[0];
            if (opcode != 0) {
                // PAKETİ BAŞARIYLA YAKALADIK!
                // SafeLogRecv(pData1, len1); diyerek işliyoruz
                return true;
            }
        }
    } __except(1) { } // Bellek okuma hatası olursa çökmeyi engelle
    return false;
}

void SafeProcessPacket(void* ecx_val, void* edx_val, void* esp4_val, void* esp8_val) {
    // Hangisinde paket olduğunu bilmiyoruz, teker teker deniyoruz
    if (TryProcessPacket(esp4_val)) return;
    if (TryProcessPacket(esp8_val)) return;
    if (TryProcessPacket(edx_val)) return;
    if (TryProcessPacket(ecx_val)) return;
   
    // Eğer hiçbirinde yoksa, This pointer'ı (ECX) içindeki tüm pointerlarda ara!
    if (!ecx_val) return;
    __try {
        for (int i = 0; i < 20; i++) {
            uint32_t val = *(uint32_t*)((uint32_t)ecx_val + (i * 4));
            if (TryProcessPacket((void*)val)) return; // Bulunursa dur
        }
    } __except(1) {}
}

static char g_mobNameResult[64];

MobInfo MobScanner::GetClosestMob(float charX, float charY, uint32_t excludeId, uint32_t minId, uint32_t maxId, float centerX, float centerZ, float maxCenterRange, const std::string& filterName) {
    auto mobs = GetMobList();
    MobInfo closest = {0};
    float minDistance = 99999.0f;
    
    // Aranan ismi küçük harfe çevirir (büyük-küçük harf duyarlılığını kaldırmak için)
    std::string lowerFilter = filterName;
    std::transform(lowerFilter.begin(), lowerFilter.end(), lowerFilter.begin(), ::tolower);
    
    for (const auto& mob : mobs) {
        if (mob.ID == excludeId) continue;
        if (minId > 0 && mob.ID < minId) continue; // Minimum ID kontrolü
        if (maxId > 0 && mob.ID > maxId) continue; // Maksimum ID kontrolü
        
        // Eğer aranan isim varsa:
        if (!filterName.empty()) {
            std::string lowerName = mob.Name;
            std::transform(lowerName.begin(), lowerName.end(), lowerName.begin(), ::tolower);
            
            // Eğer aranan filtre kelimesi mob isminde geçmiyorsa bu mobu es geç (continue)
            if (lowerName.find(lowerFilter) == std::string::npos) continue;
        } else {
            // Eğer filtre yoksa standart kontrol (NPC, Kapı veya Anıtları seçmeme mantığı vs.)
            if (mob.Nation == 0) continue;
            // ... (diğer NPC korumaları)
        }

        // Koordinat kontrolü (eğer merkez mesafe filtresi de aktifse)
        if (maxCenterRange > 0.0f) {
            float distToCenter = std::sqrt((mob.X - centerX) * (mob.X - centerX) + (mob.Y - centerZ) * (mob.Y - centerZ));
            if (distToCenter > maxCenterRange) continue;
        }

        float dist = mob.DistanceTo(charX, charY);
        if (dist < minDistance && dist <= 70.0f) { // En fazla 70 range içerisindeyse hedefe kitlenir
            minDistance = dist;
            closest = mob;
        }
    }
    return closest;
}
