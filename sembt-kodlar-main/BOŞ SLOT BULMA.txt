    // Bankadaki boş slotları bul
    inline std::vector<int> GetEmptyBankSlots()
    {
        std::vector<int> emptySlots;
       
        DWORD dlg = *(DWORD*)KO_PTR_DLG;
        if (!dlg) return emptySlots;
       
        DWORD warehouse = *(DWORD*)(dlg + KO_OFF_UIWareHouseDlg);
        if (!warehouse) return emptySlots;
       
        for (int i = 0; i < KO_BANK_TOTAL_SLOTS; ++i)
        {
            DWORD itemPtr = *(DWORD*)(warehouse + KO_OFF_UIWareHouseDlg_FirstItem + (i * 4));
            if (itemPtr == 0) {
                emptySlots.push_back(i);
            }
        }
       
        return emptySlots;
    }

    // Envanterdeki boş slotları bul (14-41 arası, 0-13 ekipman slotu)
    inline std::vector<int> GetEmptyInventorySlots()
    {
        std::vector<int> emptySlots;
       
        DWORD dlg = *(DWORD*)KO_PTR_DLG;
        if (!dlg) return emptySlots;
       
        DWORD inventory = *(DWORD*)(dlg + KO_OFF_UIInventory);
        if (!inventory) return emptySlots;
       
        // 0'dan başla (0-41 envanter slotları)
        for (int i = 0; i < 42; ++i)
        {
            DWORD itemPtr = *(DWORD*)(inventory + KO_OFF_UIInventory_FirstItem + (i * 4));
            if (itemPtr == 0) {
                emptySlots.push_back(i);
            }
        }
       
        return emptySlots;
    }

    // Boş slot string formatı
    inline std::string GetEmptyBankSlotsString()
    {
        auto slots = GetEmptyBankSlots();
        std::stringstream ss;
        for (size_t i = 0; i < slots.size(); ++i)
        {
            if (i > 0) ss << ",";
            ss << slots[i];
        }
        return ss.str();
    }

    inline std::string GetEmptyInventorySlotsString()
    {
        auto slots = GetEmptyInventorySlots();
        std::stringstream ss;
        for (size_t i = 0; i < slots.size(); ++i)
        {
            if (i > 0) ss << ",";
            ss << slots[i];
        }
        return ss.str();
